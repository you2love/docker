💾 Docker 数据卷详解

实现数据持久化，管理容器数据

命名卷 (Named Volumes)

由 Docker 管理，独立于容器的生命周期

# 创建命名卷
docker volume create mydata

# 使用命名卷运行容器
docker run -d -v mydata:/app/data nginx

# 查看卷列表
docker volume ls

# 查看卷详情
docker volume inspect mydata

# 删除卷
docker volume rm mydata

绑定挂载 (Bind Mounts)

将主机目录或文件挂载到容器

# 挂载主机目录
docker run -d -v /host/path:/container/path nginx

# 只读挂载
docker run -d -v /host/path:/container/path:ro nginx

# 挂载单个文件
docker run -d -v /host/file.txt:/container/file.txt nginx

# 开发环境示例
docker run -d -v $(pwd):/app -w /app node npm start

tmpfs 挂载

内存存储，容器停止后数据丢失

# 使用 tmpfs 挂载
docker run -d --tmpfs /tmp:rw,size=100m nginx

# 指定大小和权限
docker run -d --tmpfs /cache:rw,noexec,size=1g nginx

# 适用于临时数据、缓存、密钥

🎼 Compose 中的数据卷配置

在 docker-compose.yml 中定义和使用数据卷

version: '3.8'

services:
  db:
    image: postgres:16
    volumes:
      # 命名卷
      - db-data:/var/lib/postgresql/data
      # 绑定挂载
      - ./backups:/backups
      # 匿名卷
      - /var/lib/postgresql/backup

  app:
    image: myapp:latest
    volumes:
      # 只读挂载
      - ./config:/app/config:ro
      # 临时目录
      - type: tmpfs
        target: /tmp
        tmpfs:
          size: 100000000
          mode: 1777

volumes:
  db-data:
    driver: local
    driver_opts:
      type: none
      o: bind
      device: /data/db
