🎼 Docker Compose 实战

使用 Compose 定义和运行多容器应用

完整的 Compose 配置示例

name: myapp

services:
  web:
    build:
      context: .
      dockerfile: Dockerfile
      args:
        - NODE_ENV=production
    ports:
      - "3000:3000"
    environment:
      - NODE_ENV=production
      - DATABASE_URL=postgres://user:pass@db:5432/myapp
      - REDIS_URL=redis://redis:6379
    volumes:
      - ./logs:/app/logs
      - web-static:/app/public
    depends_on:
      db:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - frontend
      - backend
    deploy:
      resources:
        limits:
          cpus: '1.5'
          memory: 1G
        reservations:
          cpus: '0.25'
          memory: 256M
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "wget", "-q", "--spider", \
             "http://localhost:3000/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 40s

  db:
    image: postgres:16-alpine
    environment:
      POSTGRES_DB: myapp
      POSTGRES_USER: user
      POSTGRES_PASSWORD: ${DB_PASSWORD:-secure_password}
    volumes:
      - db-data:/var/lib/postgresql/data
      - ./backups:/backups
    networks:
      - backend
    deploy:
      resources:
        limits:
          cpus: '1'
          memory: 1G
    restart: unless-stopped
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U user -d myapp"]
      interval: 10s
      timeout: 5s
      retries: 5

  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes
    volumes:
      - redis-data:/data
    networks:
      - backend
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5

networks:
  frontend:
    driver: bridge
  backend:
    driver: bridge
    internal: true

volumes:
  db-data:
  redis-data:
  web-static:

常用 Compose 命令

docker compose up -d 后台启动所有服务
docker compose down 停止并删除所有资源
docker compose logs -f 查看实时日志
docker compose ps 查看服务状态
docker compose exec SERVICE sh 进入容器
docker compose build 重新构建镜像

📋 Compose 配置详解

深入了解 docker-compose.yml 的各个配置项

服务配置 (services)

  • build: 构建配置（context, dockerfile, args）
  • image: 使用现有镜像
  • ports: 端口映射
  • environment: 环境变量
  • volumes: 数据卷挂载
  • networks: 网络连接
  • depends_on: 依赖关系
  • restart: 重启策略
  • healthcheck: 健康检查
  • deploy: 资源限制

网络配置 (networks)

  • driver: 网络驱动（bridge, overlay）
  • internal: 内部网络（无外网访问）
  • ipam: IP 地址管理
  • labels: 标签元数据

数据卷配置 (volumes)

  • driver: 卷驱动（local, nfs）
  • driver_opts: 驱动选项
  • labels: 标签元数据
